def number(value, low, high):
    return type(value) in (int, float) and math.isfinite(value) and low <= value <= high

def require(ok, message):
    if not ok:
        raise ValueError(message)

def valid_date(value):
    require(isinstance(value, str), 'Datum fehlt.')
    try:
        date.fromisoformat(value)
    except ValueError:
        raise ValueError('Ungültiges Datum.') from None

def validate_test(test):
    require(isinstance(test, dict), 'Ungültiger Test.')
    valid_date(test.get('datum'))
    require(isinstance(test.get('notizen'), str) and len(test['notizen']) <= 4000, 'Ungültige Testnotizen.')
    kind = test.get('art')
    if kind == 'Sprungtest':
        for k in JUMPS:
            v = test.get(k)
            require(isinstance(v, list) and len(v) == 3 and all(number(x,0,100) for x in v), 'Drei gültige Weiten je Sprungtest erforderlich.')
        require(any(v > 0 for k in JUMPS for v in test[k]), 'Mindestens eine gemessene Weite eintragen.')
    elif kind == 'Lauftest':
        require(number(test.get('distanz'),1,3000), 'Testdistanz fehlt.')
        require(isinstance(test.get('protokoll'),str) and 1 <= len(test['protokoll'].strip()) <= 1000, 'Start, Strecke und Zeitmessung bitte beschreiben.')
        v=test.get('versuche')
        require(isinstance(v,list) and len(v)==3 and all(number(x,0,3600) for x in v) and any(x>0 for x in v), 'Mindestens eine gemessene Zeit eintragen.')
    else:
        raise ValueError('Unbekannter Testtyp.')

def test_summary(t):
    row={'Datum':t['datum'],'Test':t['art'],'Notizen':t['notizen']}
    if t['art']=='Sprungtest':
        best={k:max(t[k]) for k in JUMPS}
        row.update({JUMPS[k]+' (m)':best[k] or None for k in JUMPS})
        left,right=best['links'],best['rechts']
        both=left>0 and right>0
        row['Differenz (cm)']=round(abs(left-right)*100,1) if both else None
        row['Abweichung (%)']=round(abs(left-right)/((left+right)/2)*100,2) if both else None
        row['Größere Weite']=('gleich' if left==right else 'links' if left>right else 'rechts') if both else 'offen'
        for k in JUMPS:
            for i,v in enumerate(t[k],1):row[f'{JUMPS[k]} V{i} (m)']=v or None
    else:
        row.update({'Distanz (m)':t['distanz'],'Protokoll':t['protokoll'],'Beste Zeit (s)':min(x for x in t['versuche'] if x>0)})
        for i,v in enumerate(t['versuche'],1):row[f'Lauf V{i} (s)']=v or None
    return row

def m_training_defaults(sport, band):
    return {
        'sport': sport, 'band': band,
        'ball': {'enabled': sport in ('Fussball', 'Basketball') and band in M_BALL_EDGES,
                 'edge_m': 17.5 if (sport, band) == ('Fussball', 'U11') else M_BALL_EDGES.get(band, 24.),
                 'series': 2 if band == 'U11' else 3, 'runs': 5,
                 'run_seconds': None, 'notes': '',
                 'e2_progression': (sport, band) == ('Fussball', 'U11'),
                 'timed_distance_m': None},
        'maximal': {'enabled': False, 'edge_m': M_MAX_EDGES.get(band, (10., 25.))[0],
                    'series': 3, 'runs': 5, 'run_seconds': None, 'units': [], 'notes': ''},
    }


def validate_m_training(value, sport, band):
    if not isinstance(value, dict):
        raise ValueError('M-Lauf-Vorgaben müssen ein Objekt sein.')
    if value and (value.get('sport', sport) != sport or value.get('band', band) != band):
        raise ValueError('M-Lauf-Vorgaben gehören zu einer anderen Sportart oder Altersklasse.')
    config = m_training_defaults(sport, band)
    for kind in ('ball', 'maximal'):
        part = value.get(kind, {})
        if not isinstance(part, dict):
            raise ValueError('M-Laufform prüfen.')
        config[kind].update(deepcopy(part))
        part = config[kind]
        if type(part['enabled']) is not bool:
            raise ValueError('M-Lauf-Auswahl prüfen.')
        edge = part['edge_m']
        if type(edge) not in (int, float) or not math.isfinite(edge) or not .5 <= edge <= 100:
            raise ValueError('M-Lauf-Kantenlänge muss zwischen 0,5 und 100 m liegen.')
        if kind == 'maximal' and band in M_MAX_EDGES:
            lower, upper = M_MAX_EDGES[band]
            if not lower <= edge <= upper:
                raise ValueError(f'Maximale M-Sprints {band}: Kantenlänge {lower:g}–{upper:g} m.')
        for field in ('series', 'runs'):
            if type(part[field]) is not int or not 1 <= part[field] <= 100:
                raise ValueError('M-Lauf-Serien und Durchgänge müssen ganze Zahlen von 1 bis 100 sein.')
        seconds = part['run_seconds']
        if seconds is not None and (type(seconds) not in (int, float) or not math.isfinite(seconds) or not 0 < seconds <= 3600):
            raise ValueError('Gemessene Durchlaufzeit muss positiv sein; leer bedeutet nicht gemessen.')
        if not isinstance(part['notes'], str) or len(part['notes']) > 4000:
            raise ValueError('M-Lauf-Notizen prüfen.')
    ball = config['ball']
    if type(ball['e2_progression']) is not bool:
        raise ValueError('U11-Aufbau muss als Auswahl gespeichert sein.')
    if ball['e2_progression'] and (sport, band) != ('Fussball', 'U11'):
        raise ValueError('Der festgelegte U11-Aufbau gehört zu Fußball U11.')
    timed = ball['timed_distance_m']
    if timed is not None and (type(timed) not in (int, float) or timed not in E2_BALL_DISTANCES):
        raise ValueError('Die gemessene Ball-Durchlaufzeit einer Strecke von 70 bis 95 m zuordnen.')
    units = config['maximal']['units']
    if not isinstance(units, list) or any(type(x) is not int or not 1 <= x <= 28 for x in units) or len(units) != len(set(units)):
        raise ValueError('M-Sprint-Einheiten müssen eindeutige TE-Nummern von 1 bis 28 sein.')
    return config


def m_load_summary(part, distance_m=None):
    per_run = 4 * part['edge_m'] if distance_m is None else distance_m
    count = part['series'] * part['runs']
    seconds = part.get('run_seconds')
    if distance_m is not None and part.get('timed_distance_m') != distance_m:
        seconds = None
    return {'per_run_m': per_run, 'runs_total': count, 'total_m': per_run * count,
            'active_seconds': None if seconds is None else seconds * count}


def m_distance_text(part, distance_m=None):
    summary = m_load_summary(part, distance_m)
    geometry = (f"ca. {distance_m:g} m je Durchlauf; vier Abschnitte, rechnerisch im Mittel {distance_m / 4:g} m "
                '(Aufbau gerundet, Kurvenverlauf variabel); '
                if distance_m is not None else
                f"4 Kanten à {part['edge_m']:g} m = {summary['per_run_m']:g} m je Durchlauf; ")
    text = (geometry +
            f"{part['runs']} Durchgänge je Serie; {summary['runs_total']} Durchgänge / "
            f"{'ca. ' if distance_m is not None else ''}{summary['total_m']:g} m insgesamt")
    if summary['active_seconds'] is None:
        return text + '; Laufdauer noch nicht gemessen'
    return text + f"; Laufzeit hochgerechnet aus {part['run_seconds']:g} s je Durchlauf: {summary['active_seconds']:g} s (ohne Pausen)"

def normalize_profile(p):
    result=new_profile();result.update(deepcopy(p))
    # Legacy cycles keep their original week-based interpretation until a new cycle is chosen.
    result['plan_model']=p.get('plan_model','legacy')
    if result['plan_model']=='legacy' and 'phasensteuerung' not in p:
        result['phasensteuerung']=dict(phase_defaults(),basis=6,jumps=3,spruenge=3)
    return result

def validate(data):
    require(isinstance(data,dict) and data.get('app')==APP_ID and data.get('schema')==2,'Keine passende Skispringen-Sicherung.')
    athletes=data.get('athletes')
    require(isinstance(athletes,dict) and len(athletes)<=10000,'Ungültiger Kader.')
    for aid,p in athletes.items():
        require(isinstance(aid,str) and 1<=len(aid)<=100 and isinstance(p,dict),'Ungültiges Profil.')
        require(isinstance(p.get('name'),str) and 1<=len(p['name'].strip())<=120,'Name fehlt.')
        require(p.get('profil') in PROFILES,'Altersklasse fehlt.')
        require(type(p.get('alter')) is int and 9<=p['alter']<=40,'Alter prüfen.')
        require(number(p.get('ski_angle',6),0,20) and type(p.get('ski_group',True)) is bool,'Skisprung-Einstellungen prüfen.')
        for field,hi in [('groesse',250),('gewicht',250),('ref_30m',120),('cheer_kg',50),('vbt_val',20),('si_val',200),('main_load',500)]:
            require(number(p.get(field),0,hi),f'Ungültiger Wert: {field}.')
        require(type(p.get('aktive_woche')) is int and 1<=p['aktive_woche']<=16,'Woche prüfen.')
        require(p.get('plan_model','legacy') in ('legacy','te16'),'Planungsmodell prüfen.')
        require(type(p.get('aktive_te',1)) is int and 1<=p.get('aktive_te',1)<=16,'Aktive Trainingseinheit prüfen.')
        require(type(p.get('zyklus_einstieg',1)) is int and 1<=p.get('zyklus_einstieg',1)<=15,'Einstiegsniveau muss TE 1–15 sein.')
        require(type(p.get('frequency')) is int and p['frequency'] in [1,2],'Wochenstruktur prüfen.')
        require(type(p.get('offset_monate')) is int and p['offset_monate'] in range(-24,25,6),'Offset prüfen.')
        require(p.get('plan_basis') in PLAN_BASES and p.get('variation') in VARIATIONS,'Planbasis oder Variation prüfen.')
        require(isinstance(p.get('main_exercise'),str) and 1<=len(p['main_exercise'].strip())<=120,'Kraft-Hauptübung prüfen.')
        require(p.get('abc_rule') in ABC_RULES and type(p.get('landetechnik')) is bool,'Planungsregel prüfen.')
        for f in ['notizen','measurement_notes']:
            require(isinstance(p.get(f),str) and len(p[f])<=4000,'Notizen prüfen.')
        require(isinstance(p.get('tests'),list) and len(p['tests'])<=10000,'Testverlauf prüfen.')
        for t in p['tests']: validate_test(t)
        validate_sessions(p.get('einheitenprotokoll',{}))
        phase_validate(p.get('phasensteuerung',{}))
        validate_organization(p.get('organisation', {}))
        validate_m_training(p.get('m_training', {}), 'Skispringen', p['profil'].split('_')[1])
        require(number(p.get('folge_rate',0),0,20),'Änderungsrate prüfen.')
        require(type(p.get('abc_progression',False)) is bool,'ABC-Freigabe prüfen.')
        require(isinstance(p.get('aktiver_makrozyklus','Bestand'),str) and 1<=len(p.get('aktiver_makrozyklus','Bestand'))<=120,'Zyklus prüfen.')
        archives=p.get('makrozyklen',{})
        require(isinstance(archives,dict) and len(archives)<=100,'Zyklusarchiv prüfen.')
        for cycle, snapshot in archives.items():
            require(isinstance(cycle,str) and 1<=len(cycle)<=120 and isinstance(snapshot,dict),'Zyklusarchiv beschädigt.')
            require(not snapshot.get('makrozyklen'),'Verschachteltes Archiv.')
            probe=deepcopy(EMPTY);probe['athletes'][aid]=normalize_profile(snapshot);validate(probe)
    plans=data.get('plans')
    require(isinstance(plans,dict) and len(plans)<=10000,'Pläne prüfen.')
    for key,p in plans.items():
        require(isinstance(key,str) and len(key)<=500 and isinstance(p,dict),'Ungültiger Plan.')
        owner=p.get('owner')
        require(owner in athletes or owner in ['standard:'+x for x in PROFILES],'Plan hat keine gültige Zuordnung.')
        require(type(p.get('week')) is int and 1<=p['week']<=16 and p.get('unit') in ['Basis','TE1','TE2'],'Planungswoche oder Einheit prüfen.')
        require(key==plan_key(owner,p['week'],p['unit'],p.get('cycle','Bestand')),'Plan-Schlüssel passt nicht.')
        valid_date(p.get('datum'))
        validate_timing(p.get('timing', {}))
        require(isinstance(p.get('notizen'),str) and len(p['notizen'])<=4000,'Plannotizen prüfen.')
        require(p.get('profil') in PROFILES,'Planprofil prüfen.')
        require(isinstance(p.get('rows'),list) and 1<=len(p['rows'])<=250,'Plan benötigt 1 bis 250 Zeilen.')
        for row in p['rows']:
            require(isinstance(row,dict) and all(isinstance(row.get(c),str) and len(row[c])<=6000 for c in COLUMNS),'Planzeilen prüfen.')
    for plan in plans.values():
        require(isinstance(plan.get('cycle','Bestand'),str) and 1<=len(plan.get('cycle','Bestand'))<=120,'Planzyklus prüfen.')
        for field in ['original_rows']:
            if field in plan:
                require(isinstance(plan[field],list) and 1<=len(plan[field])<=250,'Originalplan prüfen.')
                for row in plan[field]:require(isinstance(row,dict) and all(isinstance(row.get(c),str) and len(row[c])<=6000 for c in COLUMNS),'Originalzeile prüfen.')
        history=plan.get('revisions',[])
        require(isinstance(history,list) and len(history)<=200,'Planverlauf prüfen.')
        for old in history:
            require(isinstance(old,dict) and 'revisions' not in old,'Ungültiger Planverlauf.')
            require(old.get('owner')==plan['owner'] and old.get('cycle','Bestand')==plan.get('cycle','Bestand') and old.get('week')==plan.get('week') and old.get('unit')==plan.get('unit'),'Verlauf gehört zu anderem Plan.')
            probe=deepcopy(EMPTY);probe['athletes']=athletes;probe['plans']={plan_key(old['owner'],old['week'],old['unit'],old.get('cycle','Bestand')):old}
            validate(probe)
    require(isinstance(data.get('legacy_archive'),dict),'Altarchiv prüfen.')
    return deepcopy(data)

def plan_key(owner,week,unit,cycle='Bestand'):
    base=f'{owner}|{week}|{unit}'
    return base if cycle=='Bestand' else base+'|cycle:'+json.dumps(cycle,ensure_ascii=False)

def import_data(raw):
    """Accept both released Skispringen data and the uploaded general draft; never discard old data."""
    require(isinstance(raw,dict),'Sicherung muss ein JSON-Objekt sein.')
    if raw.get('app')==APP_ID and raw.get('schema')==2:
        out=deepcopy(raw)
        require(isinstance(out.get('athletes'),dict) and isinstance(out.get('plans'),dict),'Beschädigte Sicherung.')
        for aid,old in out['athletes'].items():
            require(isinstance(old,dict),'Ungültiges Profil.')
            upgraded=normalize_profile(old)
            if 'station3' in upgraded:
                upgraded['legacy_station3']=upgraded.pop('station3')
                upgraded['exercise_review']=True
            out['athletes'][aid]=upgraded
        for plan in out['plans'].values():
            require(isinstance(plan,dict),'Ungültiger Plan.')
            if plan.get('exercise_rules_version')!=VERSION:plan['exercise_review']=True
        return validate(out)
    out=deepcopy(EMPTY)
    if raw.get('app')==APP_ID and raw.get('schema')==1:
        require(isinstance(raw.get('athletes'),dict) and isinstance(raw.get('plans'),dict),'Beschädigte ältere Sicherung.')
        for aid,old in raw['athletes'].items():
            require(isinstance(old,dict) and old.get('gruppe') in ['U16 weiblich','U18 männlich'],'Unbekannte frühere Gruppe.')
            p=normalize_profile(old)
            p['profil']='Skispringen_U17_w' if old['gruppe']=='U16 weiblich' else 'Skispringen_U20_m'
            # Preserve exact existing age and measurements. General bands are provisional.
            p['profile_review']=True
            p['source_profile']=deepcopy(old)
            out['athletes'][aid]=p
        out['legacy_archive']['schema1']=deepcopy(raw)
    elif set(raw)=={'Skispringen'} and isinstance(raw['Skispringen'],dict):
        for name,old in raw['Skispringen'].items():
            require(isinstance(name,str) and isinstance(old,dict),'Beschädigtes Profil im Entwurf.')
            old=deepcopy(old)
            old['profil']=old.get('profil') or ('Skispringen_'+old.get('ak','')+('' if old.get('ak') in ['U11','U13'] else '_w' if str(old.get('geschlecht','')).lower()=='weiblich' else '_m'))
            old['profil']=old['profil'].replace('U19 / U20','U20').replace('U23 / Aktive','U23')
            require(old.get('profil') in PROFILES,'Ungültiges Entwurfsprofil.')
            require(number(old.get('groesse'),0,2.5),'Körpergröße im Entwurf muss in Metern vorliegen.')
            old=deepcopy(old)
            old['profil']=old.get('profil') or ('Skispringen_'+old.get('ak','')+('' if old.get('ak') in ['U11','U13'] else '_w' if str(old.get('geschlecht','')).lower()=='weiblich' else '_m'))
            old['profil']=old['profil'].replace('U19 / U20','U20').replace('U23 / Aktive','U23')
            p=normalize_profile(old);p.update(name=name,groesse=round(old['groesse']*100,3),source_profile=deepcopy(old))
            # Old draft telemetry defaults do not prove a measurement exists.
            p['vbt_val']=0.;p['si_val']=0.
            if old.get('sprungtests'):
                p['tests']=[{'art':'Sprungtest','datum':t['datum'],'notizen':t.get('notizen',''),'links':t['hop_links'],'rechts':t['hop_rechts'],'schluss':t['schluss']} for t in old['sprungtests']]
            if old.get('makrozyklen'):
                # Different profile schemas: preserve complete source, avoid misinterpreting archives.
                p['makrozyklen']={}
            plan=old.get('planung',{})
            p['frequency']=plan.get('einheiten',p['frequency'])
            p['aktive_woche']=min(14,plan.get('startwoche',p['aktive_woche']))
            p['main_load']=plan.get('kreuzheben_last',p['main_load'])
            p['measurement_notes']='Alte VBT-/SI-Werte im Originalarchiv erhalten; Messung bitte bestätigen.'
            p['profile_review']=True
            p['plan_basis']=PLAN_BASES[0]
            aid=uuid.uuid5(uuid.NAMESPACE_URL,'docathletic:skispringen:'+name).hex
            out['athletes'][aid]=p
        out['legacy_archive']['draft465']=deepcopy(raw)
    else:raise ValueError('Diese Datei ist keine unterstützte Skispringen-Sicherung.')
    return validate(out)

def database_url():
    return setting('DOC_SKISPRINGEN_DATABASE_URL') or setting('DOC_ATHLETIC_DATABASE_URL')

class StorageError(OSError):
    pass

def remote_state(data=None,revision=None):
    """Transactional PostgreSQL store; strictly separate Skispringen table."""
    try:
        import psycopg
    except ImportError as exc:
        raise StorageError('Externe Datenbank: psycopg fehlt; requirements.txt aktualisieren.') from exc
    try:
        with psycopg.connect(database_url(),connect_timeout=10) as con:
            con.execute('CREATE TABLE IF NOT EXISTS doc_athletic_ski_state (id INTEGER PRIMARY KEY CHECK(id=1), revision BIGINT NOT NULL, payload TEXT NOT NULL)')
            con.execute('CREATE TABLE IF NOT EXISTS doc_athletic_ski_history (revision BIGINT PRIMARY KEY, payload TEXT NOT NULL, saved_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP)')
            initial=json.dumps(EMPTY,ensure_ascii=False,allow_nan=False)
            con.execute('INSERT INTO doc_athletic_ski_state VALUES (1,0,%s) ON CONFLICT (id) DO NOTHING',(initial,))
            rev,payload=con.execute('SELECT revision,payload FROM doc_athletic_ski_state WHERE id=1 FOR UPDATE').fetchone()
            if data is None:
                loaded=import_data(json.loads(payload))
                replacement=json.dumps(loaded,ensure_ascii=False,allow_nan=False)
                if json.loads(payload)!=loaded:
                    con.execute('INSERT INTO doc_athletic_ski_history(revision,payload) VALUES (%s,%s) ON CONFLICT (revision) DO NOTHING',(rev,payload))
                    rev+=1
                    con.execute('UPDATE doc_athletic_ski_state SET revision=%s,payload=%s WHERE id=1',(rev,replacement))
                return loaded,rev
            if rev!=revision:raise Conflict('Eine andere Sitzung hat gespeichert. Stand neu laden und abgleichen.')
            replacement=json.dumps(validate(data),ensure_ascii=False,allow_nan=False)
            if json.loads(payload)==data:return rev
            con.execute('INSERT INTO doc_athletic_ski_history(revision,payload) VALUES (%s,%s)',(rev,payload))
            con.execute('UPDATE doc_athletic_ski_state SET revision=revision+1,payload=%s WHERE id=1',(replacement,))
            return rev+1
    except psycopg.Error as exc:
        raise StorageError('Externe Skispringen-Datenbank nicht erreichbar oder nicht beschreibbar. Verbindungsdaten und Dienst prüfen; keine lokale Ersatzspeicherung erfolgt.') from exc


def load_data():
    if database_url():return remote_state()
    DATA_DIR.mkdir(parents=True,exist_ok=True)
    with closing(sqlite3.connect(DB_PATH,timeout=10)) as con,con:
        con.execute('CREATE TABLE IF NOT EXISTS state (id INTEGER PRIMARY KEY, revision INTEGER NOT NULL, payload TEXT NOT NULL)')
        con.execute('CREATE TABLE IF NOT EXISTS history (revision INTEGER PRIMARY KEY,payload TEXT NOT NULL,saved_at TEXT DEFAULT CURRENT_TIMESTAMP)')
        con.execute('BEGIN IMMEDIATE')
        row=con.execute('SELECT revision,payload FROM state WHERE id=1').fetchone()
        if row is None:
            initial=deepcopy(EMPTY)
            alternative=DATA_DIR/'doc_athletic_skispringen_v201.sqlite3'
            if alternative.exists():
                with closing(sqlite3.connect(f'file:{alternative}?mode=ro',uri=True)) as oldcon:
                    old=oldcon.execute('SELECT payload FROM state WHERE id=1').fetchone()
                    if old:initial=import_data(json.loads(old[0]))
            con.execute('INSERT INTO state VALUES (1,0,?)',(json.dumps(initial,ensure_ascii=False,allow_nan=False),))
            return initial,0
        rev,payload=row;raw=json.loads(payload);data=import_data(raw)
        if raw!=data:
            con.execute('INSERT OR IGNORE INTO history(revision,payload) VALUES (?,?)',(rev,payload))
            rev+=1
            con.execute('UPDATE state SET revision=?,payload=? WHERE id=1',(rev,json.dumps(data,ensure_ascii=False,allow_nan=False)))
        return data,rev

def save_data(data,revision):
    if database_url():return remote_state(data,revision)
    payload=json.dumps(validate(data),ensure_ascii=False,allow_nan=False)
    with closing(sqlite3.connect(DB_PATH,timeout=10)) as con,con:
        con.execute('BEGIN IMMEDIATE')
        old=con.execute('SELECT revision,payload FROM state WHERE id=1').fetchone()
        if old is None or old[0]!=revision:raise Conflict('Eine andere Sitzung hat gespeichert. Eingaben notieren, Stand neu laden und abgleichen.')
        if json.loads(old[1])==data:return revision
        con.execute('INSERT OR IGNORE INTO history(revision,payload) VALUES (?,?)',old)
        con.execute('UPDATE state SET revision=revision+1,payload=? WHERE id=1',(payload,))
        require(con.execute('SELECT payload FROM state WHERE id=1').fetchone()[0]==payload,'Speicherung nicht bestätigt.')
    return revision+1

def commit(data,message):
    try:rev=save_data(data,st.session_state.rev)
    except (ValueError,Conflict,sqlite3.Error,OSError) as exc:
        st.error(f'Nicht gespeichert: {exc}');return False
    st.session_state.data=deepcopy(data);st.session_state.rev=rev;st.session_state.notice=message
    return True

def phase_apply_rows(rows,profile,band,te):
    cfg=profile.get('phasensteuerung',{})
    planned,effective=phase_status(cfg,band,te)
    if effective=='Bestehender Plan':return rows
    if effective=='Zyklus abgeschlossen':
        return [dict(zip(COLUMNS,['Makrozyklus','Zyklus abgeschlossen','—','Neuen Zyklus planen','—','—','—','Vorhandene Pläne und Ist-Protokolle bleiben gespeichert.']))]
    # Stage only the selected strength families; leave sport-specific technique separate.
    found_front=False
    for row in rows:
        name=row['Trainingsmittel']
        if row['Block']=='03 Eigenständige Kraft-Hauptübung' and name.lower().startswith('kreuzheb'):
            kind='anreiss' if band=='U15' and profile['profil'].endswith('_w') else 'kreuzheben'
            row['Trainingsmittel']=phase_exercise_label(kind,effective)
            row['Intensität']='Technisch kontrolliert' if effective=='Grundlast' else 'Explosiv bei sauberer Technik'
            if kind=='anreiss':row['Zusatzlast']=f"Powerbag {profile['main_load']:g} kg (individuell)" if profile.get('main_load',0)>0 else 'Powerbag 10–15 kg (bestehende U15w-Vorgabe)'
            chosen=phase_load(cfg,kind,band,te)
            if chosen:row['Zusatzlast']=('Powerbag ' if kind=='anreiss' else 'Hex Bar ')+chosen
        elif 'Front Squat' in name:
            found_front=True;row['Trainingsmittel']=phase_exercise_label('front_squat',effective)
            row['Intensität']='Technisch kontrolliert' if effective=='Grundlast' else 'Explosiv bei sauberer Technik'
            chosen=phase_load(cfg,'front_squat',band,te)
            if chosen:row['Zusatzlast']='Powerbag '+chosen
    # Independent Front Squat entry: never substitute Cheerleading or Squatstoß for it.
    if not found_front and band not in ('U11','U13') and not profile.get('_ski_no_front',False):
        female=profile['profil'].endswith('_w')
        stock={'U15':('8–12','12–15'),'U17':('8–10','15'),'U20':('8–12','17'),'U23':('10–15','20'),'MASTER':('10–15','20')}
        load='Powerbag '+stock[band][0 if female else 1]+' kg'
        chosen=phase_load(cfg,'front_squat',band,te)
        if chosen:load='Powerbag '+chosen
        row=dict(zip(COLUMNS,['03 Eigenständiger Front Squat',phase_exercise_label('front_squat',effective),'3','10 Wdh. (an Ist-Verlauf anpassen)',load,'Technisch sauber','Stationswechsel bis 60 s','Grundlast und Fortschritt individuell prüfen']))
        insert=next((i for i,r in enumerate(rows) if r['Block'].startswith('03')),len(rows)-1)
        rows.insert(insert,row)
    for row in rows:
        row['Fokus']+=f' | Phase: {effective}; vorgesehen: {planned}'
    return rows


def generate_plan(profile, week, unit='Basis'):
    sport = 'Skispringen'
    frequency = profile.get('frequency', 1)
    unit = effective_unit(frequency, unit)
    # Work on copies: neither stored profiles nor source rows are mutated.
    profile = deepcopy(profile)
    profile['_te_number'] = (week-1)*frequency+(2 if unit == 'TE2' else 1)
    te=profile['_te_number']
    if profile.get('plan_model')=='te16':
        require(type(week) is int and 1<=te<=16,'Trainingseinheit außerhalb des 16-TE-Zyklus.')
        if te==16:
            return [dict(zip(COLUMNS,['Retest','Sprungtests und 30-m-Parallelstart','Nach Testprotokoll','Bekannte Testausführung wiederholen','Individuell','Vergleichbare Messbedingungen','Vollständige Erholung','Fünfer-Hop links/rechts, Fünfer-Schlusssprung und Lauftest getrennt dokumentieren. Trainer entscheidet über den nächsten Zyklus.']))]
        profile['_load_level']=min(15,profile.get('zyklus_einstieg',1)+te-1)
    rows = generate_plan_content(profile, week, unit)
    if len(rows) == 1 and rows[0]['Block'] == 'Makrozyklus': return rows
    config = validate_organization(profile.get('organisation', {}))
    if unit == 'Basis':
        second = generate_plan_content(profile, week, 'TE2')
        # Include complementary coordination/reactivity from the alternate session.
        # Do not concatenate two complete sessions (no duplicate warm-up or ABC).
        names = {r['Trainingsmittel'] for r in rows}
        additions = []
        for row in second:
            eligible = (row['Block'].startswith('03') and
                        ('Steigesprünge' in row['Trainingsmittel'] or 'Landetechnik' in row['Trainingsmittel'] or
                         profile['profil'].split('_')[1] in ('U11', 'U13'))) if sport == 'Basketball' else row['Block'].startswith('04')
            if eligible and row['Trainingsmittel'] not in names:
                additions.append(deepcopy(row)); names.add(row['Trainingsmittel'])
        insert = next((i for i,r in enumerate(rows) if 'Transfer' in r['Block'] or 'Cool-Down' in r['Block']), len(rows))
        rows[insert:insert] = additions
    if unit == 'TE1' and sport == 'Skispringen':
        # The warm-up / technique ABC stays; separate maximal sprint work moves to TE2.
        rows = [r for r in rows if r['Block'] != '03 ZNS-Beschleunigung']
    m_rows = m_training_rows(profile.get('m_training', {}), sport, profile['profil'].split('_')[1],
                             profile['_te_number'], frequency, unit == 'TE2')
    insert = next((i for i, row in enumerate(rows) if not row['Block'].startswith('01')), len(rows))
    rows[insert:insert] = m_rows
    warm, stations, transfers, cool = [], [], [], []
    for row in rows:
        is_station = ((row['Block'].startswith('03') and not any(x in row['Block'] for x in ('Transfer','Ausdauer','ZNS-Beschleunigung'))) if sport == 'Basketball' else (
                      row['Block'].startswith('04') or 'Eigenständige' in row['Block']))
        if is_station:
            row['Pause'] = 'Stationswechsel bis 60 s; Serienpause separat'
            if unit == 'TE1': row['Sätze'] = str(config.get('main_sets', 4))
            stations.append(row)
        elif 'Cool-Down' in row['Block']: cool.append(row)
        elif row['Block'].startswith(('01', '02')): warm.append(row)
        else: transfers.append(row)
    block_count = max(1, math.ceil(len(stations)/5))
    organized = list(warm)
    cursor = 0
    for block in range(block_count):
        size = len(stations)//block_count + (block < len(stations)%block_count)
        for index, row in enumerate(stations[cursor:cursor+size], 1):
            original_block = row['Block']
            row['Block'] = f'Komplexblock {block+1} · Station {index}'
            row['Fokus'] = original_block + ' | ' + row['Fokus']
            organized.append(row)
        cursor += size
        # Preserve each complete transfer prescription exactly once.
        left = block*len(transfers)//block_count
        right = (block+1)*len(transfers)//block_count
        for row in transfers[left:right]:
            row['Block'] = f'Nach Block {block+1}: Transfer'
            organized.append(row)
    return organized+cool

def profile_signature(profile):
    return hashlib.sha256(json.dumps({k:v for k,v in profile.items() if k not in ['tests','source_profile','einheitenprotokoll','makrozyklen','folge_rate']},sort_keys=True,ensure_ascii=False).encode()).hexdigest()

def validate_timing(timing):
    if not isinstance(timing, dict):
        raise ValueError('Ungültige Zeitplanung.')
    fields = ('planned_athletic_min', 'actual_athletic_min', 'planned_transfer_min', 'actual_transfer_min')
    for field in fields:
        value = timing.get(field)
        if value is not None and (type(value) not in (int, float) or not math.isfinite(value) or not 0 <= value <= 1440):
            raise ValueError('Dauer muss eine endliche, nicht negative Minutenzahl sein.')
    for key in ('athletic_hint', 'transfer_hint', 'model'):
        if key in timing and (not isinstance(timing[key], str) or len(timing[key]) > 1500):
            raise ValueError('Ungültiger Hinweis zur Zeitplanung.')
    return deepcopy(timing)

def validate_organization(config):
    if not isinstance(config, dict):
        raise ValueError('Ungültige Blockorganisation.')
    for key, low, high in [('group_size', 3, 5), ('main_sets', 4, 5), ('gap_days', 1, 7)]:
        value = config.get(key, {'group_size': 4, 'main_sets': 4, 'gap_days': 3}[key])
        if type(value) is not int or not low <= value <= high:
            raise ValueError('Ungültige Blockorganisation: ' + key)
    return deepcopy(config)

def effective_unit(frequency, unit):
    # A single session always contains the combined programme, even with an old TE2 selection.
    if frequency not in (1, 2):
        raise ValueError('Eine oder zwei zusätzliche Athletikeinheiten auswählen.')
    return 'Basis' if frequency == 1 else ('TE2' if unit == 'TE2' else 'TE1')

def weekly_timing(sport, band, frequency, unit):
    unit = effective_unit(frequency, unit)
    timing = dict(planned_athletic_min=None, planned_transfer_min=None,
                  actual_athletic_min=None, actual_transfer_min=None)
    timing['model'] = ('1 zusätzliche Athletikeinheit pro Woche: vollständiges Komplextraining'
                       if frequency == 1 else '2 zusätzliche Athletikeinheiten pro Woche: ' +
                       ('Haupteinheit' if unit == 'TE1' else 'kürzere zweite Einheit'))
    timing['athletic_hint'] = 'Dauer nach vollständigem Einheitsplan; Erwärmung eingeschlossen.'
    timing['transfer_hint'] = 'Sportartspezifischer Anschluss wird separat geplant.'
    if frequency == 2 and sport != 'Skispringen':
        main = {'U11': (60, 60), 'U13': (70, 70), 'U15': (75, 75), 'U17': (75, 75)}
        second = {'U11': (35, 35), 'U13': (40, 40), 'U15': (45, 45), 'U17': (45, 45)}
        low, high = (main.get(band, (80, 90)) if unit == 'TE1' else second.get(band, (50, 60)))
        timing['planned_athletic_min'] = float(low)
        label = str(low) if low == high else f'{low}–{high}'
        timing['athletic_hint'] = f'Richtwert {label} Minuten einschließlich Erwärmung; individuell veränderbar.'
        if sport in ('Fussball', 'Basketball'):
            timing['planned_transfer_min'] = 30.
            timing['transfer_hint'] = ('Fußballspezifischer Anschluss: Richtwert 30–45 Minuten zusätzlich.'
                                       if sport == 'Fussball' else
                                       'Basketballspezifischer Anschluss: Vorschlag 30–45 Minuten zusätzlich.')
    if sport == 'Leichtathletik':
        timing['transfer_hint'] = 'Anschluss beispielsweise lockere Steigerungsläufe; keine pauschale Zusatzdauer.'
    if sport == 'Skispringen':
        timing['athletic_hint'] = 'Sportartspezifische Dauer einschließlich Erwärmung eintragen.'
        timing['transfer_hint'] = 'Skisprungspezifischer Transfer im eigenen Ablauf; kein pauschaler Ballteil.'
    return timing

def session_record(record, cycle, te, unit):
    sessions = record.get('einheitenprotokoll', {})
    specific = sessions.get(json.dumps([cycle, te, unit], ensure_ascii=False))
    if specific is not None: return specific
    legacy = sessions.get(json.dumps([cycle, te], ensure_ascii=False))
    if legacy is None: return None
    legacy_unit = legacy.get('unit')
    if legacy_unit is None:
        # Old exports recorded the unit in their heading. Keep unmatched legacy entries
        # intact instead of attaching the same Ist record to two different weekly models.
        for candidate in ('Basis', 'TE1', 'TE2'):
            if f'· {candidate} ·' in legacy.get('plan', ''):
                legacy_unit = candidate
                break
    return legacy if legacy_unit == unit else None


def validate_sessions(sessions):
    if not isinstance(sessions, dict) or len(sessions) > 2000:
        raise ValueError("Ungültiges Einheitenprotokoll.")
    def validate_payload(item):
        if not isinstance(item, dict):
            raise ValueError("Ungültiger Einheitenstand.")
        for k in ("original_plan", "plan", "source", "cycle", "notes", "saved_at"):
            if not isinstance(item.get(k, ""), str) or len(item.get(k, "")) > 100000:
                raise ValueError("Ungültiger Text im Einheitenprotokoll.")
        if type(item.get("te")) is not int or not 1 <= item["te"] <= 32:
            raise ValueError("Ungültige Einheitsnummer.")
        if "performed" in item:
            try:
                date.fromisoformat(item["performed"])
            except (ValueError, TypeError):
                raise ValueError("Ungültiges Durchführungsdatum.") from None
        if "unit" in item and item["unit"] not in ("Basis", "TE1", "TE2"):
            raise ValueError("Ungültiges Wochenmodell.")
        validate_timing(item.get("timing", {}))
        rows = item.get("actual", [])
        if not isinstance(rows, list) or len(rows) > 150:
            raise ValueError("Ungültige Ist-Tabelle.")
        for row in rows:
            if not isinstance(row, dict) or set(row) != set(PROTOCOL_COLUMNS):
                raise ValueError("Ungültige Ist-Zeile.")
            if not isinstance(row["Übung"], str) or not row["Übung"].strip() or len(row["Übung"]) > 300:
                raise ValueError("Jede Ist-Zeile benötigt einen Übungsnamen.")
            for key in NUMERIC_COLUMNS:
                val = row[key]
                if val is not None and (type(val) not in (int, float) or not math.isfinite(val) or not 0 <= val <= 100000):
                    raise ValueError(f"Ungültiger Ist-Wert: {key}.")
            for key in ("Sätze", "Wdh. je Satz/Seite"):
                if row[key] is not None and int(row[key]) != row[key]:
                    raise ValueError("Sätze und Wiederholungen müssen ganze Zahlen sein.")
            if row["Technik"] not in ("Nicht bewertet", "Sauber", "Unsicher") or row["Belastung"] not in ("Nicht bewertet", "Gut bewältigt", "Grenzwertig", "Nicht bewältigt"):
                raise ValueError("Ungültige Rückmeldung.")
            if not isinstance(row["Anmerkung"], str) or len(row["Anmerkung"]) > 4000:
                raise ValueError("Ungültige Ist-Anmerkung.")
    for key, item in sessions.items():
        if not isinstance(key, str) or len(key) > 300:
            raise ValueError("Ungültiger Einheitenschlüssel.")
        validate_payload(item)
        identity = [item["cycle"], item["te"]] + ([item["unit"]] if "unit" in item else [])
        if key != json.dumps(identity,ensure_ascii=False):
            raise ValueError("Einheitenschlüssel passt nicht.")
        revisions = item.get("revisions", [])
        if not isinstance(revisions, list) or len(revisions) > 200:
            raise ValueError("Zu viele Korrekturen für diese Einheit.")
        for old in revisions:
            if "revisions" in old:
                raise ValueError("Verschachtelte Korrekturhistorie ist ungültig.")
            validate_payload(old)
            if (old.get("cycle"),old.get("te"),old.get("unit"))!=(item.get("cycle"),item.get("te"),item.get("unit")):
                raise ValueError("Korrekturverlauf gehört zu einer anderen Einheit.")

def save_unit_record(record, cycle, te, plan, source, actual=None, notes=None, performed=None, timing=None, unit=None):
    updated = deepcopy(record)
    sessions = updated.setdefault("einheitenprotokoll", {})
    key = json.dumps([cycle, te, unit] if unit is not None else [cycle, te], ensure_ascii=False)
    old = session_record(updated, cycle, te, unit) if unit is not None else sessions.get(key)
    history = deepcopy(old.get("revisions", [])) if old else []
    if old:
        previous = deepcopy(old)
        previous.pop("revisions", None)
        history.append(previous)
    item = deepcopy(old) if old else {"cycle": cycle, "te": te, "original_plan": plan, "actual": [], "notes": ""}
    item.update(plan=plan, source=source, saved_at=datetime.now(timezone.utc).isoformat(), revisions=history)
    if actual is not None:
        item["actual"] = deepcopy(actual)
    if notes is not None:
        item["notes"] = notes
    if performed is not None:
        date.fromisoformat(performed)
        item["performed"] = performed
    if unit is not None:
        if unit not in ("Basis", "TE1", "TE2"):
            raise ValueError("Ungültiges Wochenmodell.")
        item["unit"] = unit
    if timing is not None:
        item["timing"] = validate_timing(timing)
    sessions[key] = item
    validate_sessions(sessions)
    return updated

def next_recommendations(record, cycle, te, metric, percent):
    # Zeitangaben sind ausdrücklich keine Last-/Wiederholungsprogression.
    if metric not in ("Last (kg)", "Wdh. je Satz/Seite", "Strecke (m)") or not 0 <= percent <= 20:
        raise ValueError("Ungültige Steigerungseinstellung.")
    latest = {}
    entries = sorted(record.get("einheitenprotokoll", {}).values(), key=lambda x: (x.get("performed", ""), x.get("saved_at", "")))
    for item in entries:
        if item["cycle"] != cycle or item["te"] >= te:
            continue
        grouped = {}
        for row in item.get("actual", []):
            identity = row["Übung"].strip().casefold()
            grouped.setdefault(identity, []).append(row)
        for identity, rows in grouped.items():
            candidates = [r for r in rows if r[metric] is not None and r[metric] > 0]
            if not candidates:
                continue
            combined = deepcopy(min(candidates, key=lambda r: r[metric]))
            combined["Technik"] = "Unsicher" if any(r["Technik"] == "Unsicher" for r in rows) else "Sauber" if all(r["Technik"] == "Sauber" for r in rows) else "Nicht bewertet"
            combined["Belastung"] = "Nicht bewältigt" if any(r["Belastung"] == "Nicht bewältigt" for r in rows) else "Grenzwertig" if any(r["Belastung"] == "Grenzwertig" for r in rows) else "Gut bewältigt" if all(r["Belastung"] == "Gut bewältigt" for r in rows) else "Nicht bewertet"
            latest[identity] = (combined, item)
    output = []
    for row, item in latest.values():
        value = row[metric]
        if value is None or value <= 0:
            continue
        good = row["Technik"] == "Sauber" and row["Belastung"] == "Gut bewältigt"
        bad = row["Technik"] == "Unsicher" or row["Belastung"] in ("Grenzwertig", "Nicht bewältigt")
        factor = -percent if bad else percent if good else 0
        target = value * (1 + factor / 100)
        if metric == "Wdh. je Satz/Seite":
            target = max(1, math.floor(target))
        else:
            target = round(target, 2)
        reason = "Reduktionsvorschlag aus Rückmeldung" if bad else "Steigerungsvorschlag aus Rückmeldung" if good else "Beibehalten; Rückmeldung unvollständig"
        if percent == 0:
            reason = "Beibehalten; Änderungsrate 0 %"
        effective = (target/value-1)*100
        output.append({"Übung":row["Übung"], "Ausgangswert":value, "Parameter":metric, "Empfehlung":target,
                       "Änderung tatsächlich (%)":round(effective,2), "Begründung":reason,
                       "Quelle":f'{item["cycle"]}, TE {item["te"]}'})
    return output

class PlanTextParser(HTMLParser):
    def __init__(self):
        super().__init__(); self.parts=[]; self.hidden=0
    def handle_starttag(self, tag, attrs):
        if tag in ("style","script","title"):self.hidden+=1
        if tag in ("tr", "p", "h3", "br"):
            self.parts.append("\n")
    def handle_endtag(self, tag):
        if tag in ("style","script","title"):self.hidden=max(0,self.hidden-1)
        if tag in ("td", "th"):
            self.parts.append(" | ")
    def handle_data(self, data):
        if not self.hidden:self.parts.append(data)

def plan_as_text(html):
    parser=PlanTextParser();parser.feed(html)
    return "\n".join(line.strip() for line in "".join(parser.parts).splitlines() if line.strip())

def protocol_rows(frame):
    rows=[]
    for row in frame.to_dict("records"):
        if not str(row.get("Übung") or "").strip():
            if all(pd.isna(row.get(k)) for k in NUMERIC_COLUMNS):
                continue
            raise ValueError("Bitte den Übungsnamen zu den eingetragenen Werten ergänzen.")
        if all(pd.isna(row.get(k)) for k in NUMERIC_COLUMNS) and row.get("Technik")=="Nicht bewertet" and row.get("Belastung")=="Nicht bewertet" and (pd.isna(row.get("Anmerkung")) or not str(row.get("Anmerkung") or "").strip()):
            continue
        clean={k:row.get(k) for k in PROTOCOL_COLUMNS}
        for k in NUMERIC_COLUMNS:
            clean[k]=None if pd.isna(clean[k]) else float(clean[k])
        for k in ("Übung","Anmerkung"):
            clean[k]="" if pd.isna(clean[k]) else str(clean[k]).strip()
        rows.append(clean)
    return rows



def preserve_plan_revision(plan,old):
    plan=deepcopy(plan)
    plan['original_rows']=deepcopy(old.get('original_rows',old['rows']) if old else plan['rows'])
    history=deepcopy(old.get('revisions',[])) if old else []
    if old:
        previous=deepcopy(old);previous.pop('revisions',None);history.append(previous)
    plan['revisions']=history
    plan['saved_at']=datetime.now(timezone.utc).isoformat()
    return plan

def cycle_change(data,aid,name,create=True,entry_level=None):
    require(isinstance(name,str),'Zyklusnamen eingeben.')
    name=name.strip()
    if create:require(type(entry_level) is int and 1<=entry_level<=15,'Einstiegsniveau ausdrücklich wählen: TE 1–15.')
    out=deepcopy(data);p=out['athletes'][aid]
    current=p.get('aktiver_makrozyklus','Bestand');archives=deepcopy(p.get('makrozyklen',{}))
    require(isinstance(name,str) and 1<=len(name.strip())<=120 and name!=current,'Gültigen anderen Zyklusnamen eingeben.')
    require(name not in archives if create else name in archives,'Zyklusname bereits vorhanden oder unbekannt.')
    snapshot={k:deepcopy(v) for k,v in p.items() if k not in ['makrozyklen','einheitenprotokoll','tests','aktiver_makrozyklus']}
    if not create:
        restored=deepcopy(archives.pop(name))
        if restored.get('profil')!=p.get('profil'):
            for k in ('main_exercise','main_load','phasensteuerung','m_training','landetechnik','ski_angle'):
                if k in p:restored[k]=deepcopy(p[k])
        # Identity, measurements and full ledger remain current when changing cycles.
        for k in ['name','alter','profil','groesse','gewicht','tests','einheitenprotokoll','measurement_notes','vbt_val','si_val','ref_30m']:
            restored[k]=deepcopy(p.get(k,new_profile().get(k)))
        p=normalize_profile(restored)
    else:
        p.update(aktive_woche=1,aktive_te=1,abc_progression=True,plan_model='te16',zyklus_einstieg=entry_level)
        p['phasensteuerung']=phase_defaults()
    archives[current]=snapshot;p['makrozyklen']=archives;p['aktiver_makrozyklus']=name
    out['athletes'][aid]=p
    return validate(out)

# Gemeinsame Planungsregeln von Frank Müller, Stand 20.09.2026.
# Nur Planvorlagen: keine Messsperre, keine automatische Geräteansteuerung.
PHASE_EXERCISES={'front_squat':'Front Squat','kreuzheben':'Kreuzheben','anreiss':'Anreißen'}
PHASE_BAGS=(5,8,10,12,15,17,20)

def phase_defaults():
    return {'enabled':True,'basis':6,'jumps':5,'spruenge':5,'jumps_ready':False,'spruenge_ready':False,'references':{}}

def phase_validate(config):
    if not isinstance(config,dict):raise ValueError('Phasenplanung muss ein Objekt sein.')
    for k in ['enabled','jumps_ready','spruenge_ready']:
        if type(config.get(k,phase_defaults()[k])) is not bool:raise ValueError('Ungültige Phasenfreigabe.')
    for k in ['basis','jumps','spruenge']:
        if type(config.get(k,phase_defaults()[k])) is not int or not 1<=config.get(k,phase_defaults()[k])<=14:raise ValueError('Je Phase 1–14 Einheiten eintragen.')
    if sum(config.get(k,phase_defaults()[k]) for k in ['basis','jumps','spruenge'])>28:raise ValueError('Höchstens 28 Einheiten je Makrozyklus.')
    refs=config.get('references',{})
    if not isinstance(refs,dict) or any(k not in PHASE_EXERCISES for k in refs):raise ValueError('Übungsreferenz unbekannt.')
    for kind,ref in refs.items():
        if not isinstance(ref,dict):raise ValueError('Ungültige Übungsreferenz.')
        for k,lo,hi,default in [('kg',0,500,0),('lower',0,100,25),('upper',0,100,35),('chosen',0,500,0)]:
            x=ref.get(k,default)
            if type(x) not in (int,float) or not math.isfinite(x) or not lo<=x<=hi:raise ValueError('Ungültiger Referenzwert: '+k)
        if ref.get('lower',25)>ref.get('upper',35):raise ValueError('Prozentbereich vertauscht.')
        for k in ['confirmed','apply']:
            if type(ref.get(k,False)) is not bool:raise ValueError('Referenzfreigabe prüfen.')
        if not isinstance(ref.get('source',''),str) or len(ref.get('source',''))>4000:raise ValueError('Referenzbeschreibung prüfen.')
        if ref.get('date'):
            try:date.fromisoformat(ref['date'])
            except (ValueError,TypeError):raise ValueError('Referenzdatum prüfen.') from None
        if ref.get('confirmed') and (ref.get('kg',0)<=0 or not ref.get('date') or not ref.get('source','').strip()):raise ValueError('Bestätigte Referenz benötigt Last, Datum und Übungsausführung.')
        if kind in ('front_squat','anreiss') and ref.get('chosen',0) not in (0,)+PHASE_BAGS:raise ValueError('Powerbag nicht im Bestand.')
    return deepcopy(config)

def phase_status(config,band,te):
    cfg=phase_defaults();cfg.update(config or {})
    try:phase_validate(cfg)
    except ValueError:return 'Eingaben prüfen','Grundlast'
    if not cfg['enabled'] or band in ('U11','U13'):return 'Bestehender Plan','Bestehender Plan'
    if te>cfg['basis']+cfg['jumps']+cfg['spruenge']:return 'Zyklus abgeschlossen','Zyklus abgeschlossen'
    planned='Grundlast' if te<=cfg['basis'] else 'Jumps' if te<=cfg['basis']+cfg['jumps'] else 'Sprünge'
    if planned=='Grundlast':return planned,planned
    if not cfg['jumps_ready']:return planned,'Grundlast'
    if planned=='Jumps' or band=='U15':return planned,'Jumps'
    return planned,'Sprünge' if cfg['spruenge_ready'] else 'Jumps'

def phase_load(config,kind,band,te):
    _,effective=phase_status(config,band,te)
    ref=config.get('references',{}).get(kind,{})
    if effective not in ('Jumps','Sprünge') or not ref.get('confirmed') or not ref.get('apply'):return None
    lo=ref['kg']*ref.get('lower',25)/100;hi=ref['kg']*ref.get('upper',35)/100;chosen=ref.get('chosen',0)
    if chosen<=0 or not lo-1e-9<=chosen<=hi+1e-9:return None
    if kind in ('front_squat','anreiss') and chosen not in PHASE_BAGS:return None
    return f"{chosen:g} kg (gewählt aus {ref.get('lower',25):g}–{ref.get('upper',35):g} % der technisch sauberen {PHASE_EXERCISES[kind]}-Referenz {ref['kg']:g} kg)"

def phase_exercise_label(kind,effective):
    if effective=='Grundlast':return {'front_squat':'Front Squat – Grundlast und Technik','kreuzheben':'Kreuzheben – Grundlast und Technik','anreiss':'Anreiß-Streckung – Grundlast und Technik'}[kind]
    return PHASE_EXERCISES[kind]+(' Jumps' if effective=='Jumps' else ' Sprünge')

def get_skisprung_hardware_limits(ak: str, offset: int, gender: str, w_nr: int):
    # Basiswerte nach C2-, C1- und Bogner-Empirie
    if ak == "U11":
        s_last = 0.0
        h_last = 2.0
        pb_last = 0.0
        hocker_cm = "30 cm"
        huerde_cm = "20 cm"
    elif ak == "U13":
        # C2-Normative (13 J.)
        s_last = 4.0 if w_nr <= 4 else 6.0 if w_nr <= 8 else 8.0 if w_nr <= 12 else 10.0
        h_last = 3.0
        pb_last = 4.0 if w_nr <= 6 else 6.0
        hocker_cm = "30-40 cm"
        huerde_cm = "25 cm"
    elif ak == "U15":
        # C1-Normative (14 J.) - Strikte Geschlechtertrennung
        if gender == "weiblich":
            s_last = 5.0 if w_nr <= 4 else 7.0 if w_nr <= 8 else 8.0 if w_nr <= 12 else 10.0
            h_last = 4.0
            pb_last = 5.0 if w_nr <= 6 else 7.0
            hocker_cm = "40 cm"
            huerde_cm = "30-40 cm"
        else:
            s_last = 15.0 if w_nr <= 2 else 20.0 if w_nr <= 6 else 25.0 if w_nr <= 10 else 30.0
            h_last = 5.0
            pb_last = 10.0 if w_nr <= 6 else 15.0
            hocker_cm = "45-50 cm"
            huerde_cm = "45-60 cm"
    elif ak == "U17":
        s_last = 15.0 if gender == "weiblich" else 35.0
        h_last = 6.0
        pb_last = 12.0 if gender == "weiblich" else 16.0
        hocker_cm = "50 cm"
        huerde_cm = "50 cm"
    elif ak in ["U19 / U20", "U20"]:
        s_last = 20.0 if gender == "weiblich" else 45.0
        h_last = 8.0
        pb_last = 15.0 if gender == "weiblich" else 20.0
        hocker_cm = "55 cm"
        huerde_cm = "55 cm"
    else:
        # U23 / Aktive (Herren: Bogner-Normative)
        s_last = 25.0 if gender == "weiblich" else 55.0
        h_last = 10.0
        pb_last = 15.0 if gender == "weiblich" else 20.0
        hocker_cm = "60 cm"
        huerde_cm = "55-60 cm"

    if offset < 0 and ak not in ["U11"]:
        s_last = max(0.0, s_last - 2.5)
        pb_last = max(0.0, pb_last - 2.5)
    elif offset > 0 and ak not in ["U11", "U13"]:
        s_last += 2.5

    return s_last, h_last, pb_last, hocker_cm, huerde_cm

def generate_plan_content(profile,week,unit='Basis'):
    rows=ski_rows(profile,week,unit)
    band=profile['profil'].split('_')[1]
    female=profile['profil'].endswith('_w')
    te=profile.get('_te_number', (week-1)*profile.get('frequency',1)+(2 if unit=='TE2' else 1))
    cfg=profile.get('phasensteuerung',{})
    phase_te=min(te,sum(cfg.get(k,phase_defaults()[k]) for k in ('basis','jumps','spruenge'))) if profile.get('plan_model')=='te16' else te
    planned,effective=phase_status(cfg,band,phase_te)
    for r in rows:
        if r['Block'].startswith('04'):r['Pause']='Stationswechsel bis 60 s'
        if any(x in r['Fokus'] for x in ['IIx','Bogner','VKB','Titin','Beseitigung']):r['Fokus']='Technik, Koordination und kontrollierte Kraftübertragung'
        if 'Titin' in r['Trainingsmittel']:r['Trainingsmittel']=r['Trainingsmittel'].replace('Titin-Detonisierung','Tonusregulation')
        if 'Jumper' in r['Trainingsmittel'] and effective in ('Grundlast','Jumps','Sprünge'):
            r['Trainingsmittel']+=' · '+effective
            if effective=='Grundlast':r['Intensität']='Technisch kontrolliert, Grundlast erarbeiten'
    if band not in ('U11','U13') and unit!='TE2':
        limits={'U15':('10–15','30–45'),'U17':('30–40','40–60'),'U20':('40–50','50–70'),'U23':('40–60','60–85'),'MASTER':('40–60','60–85')}
        load=f"{profile['main_load']:g} kg" if profile.get('main_load',0)>0 else ('Powerbag ' if band=='U15' and female else 'Hex Bar ')+limits[band][0 if female else 1]+' kg'
        row=dict(zip(COLUMNS,['03 Eigenständige Kraft-Hauptübung',profile.get('main_exercise','Kreuzheben'),'3','8–12 Wdh. (an Ist-Verlauf anpassen)',load,'Technisch sauber','Stationswechsel bis 60 s','Eigenständige Kraftübung; keine Kopplung an Variationsübungen']))
        rows.insert(next((i for i,r in enumerate(rows) if r['Block'].startswith('04')),len(rows)-1),row)
    return phase_apply_rows(rows,profile,band,phase_te)

def tempo_rows(ref,model=False):
    require(number(ref,0,120),'30-m-Referenz prüfen.')
    rows=[]
    for distance,base in [(20,3.10),(30,4.20),(50,6.60),(75,9.85),(100,13.10),(150,20.20)]:
        value=ref if distance==30 and ref>0 else base*ref/4.20 if model and ref>0 else None
        r={'Distanz (m)':distance,'Quelle':'eingetragene Referenz' if distance==30 and value else 'Quellenmodell' if value else 'fehlt'}
        for pct in [100,95,90,85,80,75,70,65,60]:r[f'{pct}%']=format_tempo_time(value/(pct/100)) if value else '—'
        rows.append(r)
    return rows

SKI_IMPORT_COLUMNS = ['Name','Alter (Jahre)','Geschlecht','Größe (cm)','Gewicht (kg)','TE pro Woche','30 m (s)','Notiz']

def ski_import_rows(table):
    aliases={header_key(k):k for k in SKI_IMPORT_COLUMNS}
    for old,new in {'Alter':'Alter (Jahre)','Körpergröße (cm)':'Größe (cm)','Körperlänge (cm)':'Größe (cm)',
                    'Körpergewicht (kg)':'Gewicht (kg)','Geschlecht (w/m)':'Geschlecht',
                    '30m Referenz (s)':'30 m (s)','Notizen':'Notiz'}.items():
        aliases[header_key(old)]=new
    header=next((i for i,row in enumerate(table[:10]) if any(header_key(v)=='name' for v in row)),None)
    require(header is not None,'Keine Namensspalte gefunden. Benötigt werden Name, Alter (Jahre) und Geschlecht; Größe (cm), Gewicht (kg), TE pro Woche, 30 m (s) und Notiz sind optional.')
    raw=list(table[header])
    while raw and raw[-1] in (None,''):raw.pop()
    columns=[aliases.get(header_key(v)) for v in raw]
    require(None not in columns,'Nicht zugeordnete Spalten: '+', '.join(str(v) for v,c in zip(raw,columns) if c is None)+'. Bitte eine Tabelle mit den oben genannten Spalten verwenden.')
    require(len(set(columns))==len(columns),'Doppelte Spaltenüberschriften. Bitte jede Angabe einmal führen.')
    rows=[]
    for line,values in enumerate(table[header+1:],header+2):
        if not any(v not in (None,'') for v in values):continue
        require(not any(v not in (None,'') for v in values[len(columns):]),f'Zeile {line}: Werte ohne Spaltenüberschrift.')
        require(not any(isinstance(v,dict) or isinstance(v,str) and v.startswith('=') for v in values),f'Zeile {line}: Bitte eingetragene Werte statt Formeln verwenden.')
        values={k:('' if v is None else str(v).strip()) for k,v in zip(columns,values)}
        rows.append({key:values.get(key,'') for key in SKI_IMPORT_COLUMNS})
    require(0<len(rows)<=1000,'Bitte 1 bis 1000 Personen in einer Tabelle verwenden.')
    return rows

def ski_import_candidate(current, rows):
    import unicodedata
    def name_key(name):return ' '.join(unicodedata.normalize('NFKC',name).casefold().split())
    def numeric(raw,label,low,high,whole=False):
        import re
        text=str(raw).strip().replace(',','.')
        require(bool(re.fullmatch(r'\d+(?:\.\d+)?',text)),label+': Bitte eine Zahl eintragen.')
        value=float(text)
        require(math.isfinite(value) and low<=value<=high and (not whole or value==int(value)),label+': Wert außerhalb des zulässigen Bereichs.')
        return int(value) if whole else value
    candidate=deepcopy(current);seen=set();changes=0
    for index,row in enumerate(rows,1):
        name=str(row.get('Name','')).strip()
        require(0<len(name)<=120,f'Zeile {index}: Name fehlt oder ist zu lang.')
        key=name_key(name)
        require(key not in seen,f'Zeile {index}: Name mehrfach enthalten: {name}')
        seen.add(key)
        matches=[aid for aid,p in current['athletes'].items() if name_key(p['name'])==key]
        require(len(matches)<=1,f'Zeile {index}: Name ist im bisherigen Kader mehrdeutig: {name}')
        old=current['athletes'][matches[0]] if matches else None
        profile=deepcopy(old) if old else new_profile()
        if not old:profile['name']=name
        provided={}
        for label,field,low,high,whole in [('Alter (Jahre)','alter',9,40,True),('Größe (cm)','groesse',0,250,False),('Gewicht (kg)','gewicht',0,250,False),('TE pro Woche','frequency',1,2,True),('30 m (s)','ref_30m',0,120,False)]:
            if str(row.get(label,'')).strip():provided[field]=numeric(row[label],name+' / '+label,low,high,whole)
        require(old is not None or 'alter' in provided,f'{name}: Alter fehlt.')
        gender=str(row.get('Geschlecht','')).strip().casefold()
        require(not gender or gender in ('w','weiblich','m','männlich','maennlich'),f'{name}: Geschlecht bitte w oder m.')
        require(old is not None or bool(gender),f'{name}: Geschlecht fehlt.')
        if gender:provided['geschlecht']='Weiblich' if gender in ('w','weiblich') else 'Männlich'
        if str(row.get('Notiz','')).strip():provided['notizen']=str(row['Notiz']).strip()
        if old:
            conflicts=[field for field,value in provided.items() if old.get(field) not in (None,'',0,0.) and old.get(field)!=value]
            if gender and old['profil'].endswith(('_w','_m')):
                require(old['profil'].endswith('_w' if provided['geschlecht']=='Weiblich' else '_m'),f'{name}: Geschlecht weicht vom gespeicherten Profil ab.')
            require(not conflicts,f'{name}: Abweichende vorhandene Angaben ({", ".join(conflicts)}). Bitte im Athletenprofil korrigieren oder eine vollständige Sicherung verwenden.')
        profile.update(provided)
        if not old:
            age=profile['alter']
            band=next((b for upper,b in [(10,'U11'),(12,'U13'),(14,'U15'),(16,'U17'),(19,'U20'),(22,'U23')] if age<=upper),'MASTER')
            profile['profil']='Skispringen_'+band+('' if band in ('U11','U13') else '_w' if provided['geschlecht']=='Weiblich' else '_m')
        aid=matches[0] if matches else uuid.uuid5(uuid.NAMESPACE_URL,'docathletic:skispringen:table:'+key).hex
        require(aid not in candidate['athletes'] or bool(matches),f'{name}: Personenkennung ist bereits vergeben.')
        if profile!=old:changes+=1
        candidate['athletes'][aid]=profile
    return validate(candidate),changes

def render_import():
    st.subheader('Kaderdaten hochladen')
    st.caption(UPLOAD_HELP)

    if upload is None:return
    raw=upload.getvalue()
    try:
        kind=upload_kind(raw,upload.name)
        if kind=='backup':render_uploaded_backup(raw);return
        rows=ski_import_rows(read_table_cells(raw,upload.name))
    except Exception as exc:
        st.error('Datei nicht übernommen: '+str(exc));return
    st.caption('Pflichtangaben für neue Personen: Name, Alter und Geschlecht. Weitere Angaben können in der Vorschau ergänzt werden. Fehlende Messwerte bleiben offen; vorhandene Trainingsverläufe bleiben erhalten.')

    key='ski_upload_'+hashlib.sha256(raw).hexdigest()[:16]+'_'+str(st.session_state.rev)
    edited=st.data_editor(pd.DataFrame(rows).fillna('').astype(str),hide_index=True,num_rows='fixed',width='stretch',key=key)

    if changes==0:st.success('Diese Angaben sind bereits im Kader gespeichert.')
    if st.button('Kader übernehmen',type='primary',disabled=not changes):

            st.session_state.profile_epoch=st.session_state.get('profile_epoch',0)+1
